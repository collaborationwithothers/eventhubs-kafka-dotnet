          set -euo pipefail
          python3 - <<'PYTHON'
          import importlib.util, io, ipaddress, json, os, pathlib, re, socket, subprocess, sys, time
          path = pathlib.Path('infra/scripts/phase/phase.py')
          sys.path.insert(0, str(path.parent.resolve()))
          spec = importlib.util.spec_from_file_location('smoke_phase', path)
          module = importlib.util.module_from_spec(spec)
          spec.loader.exec_module(module)
          phase = module.Phase()
          output = phase.read('ephemeral-output.json')
          values = phase.values('start-line', 1)['runtime']
          assert values['eventhubs'] == module.unwrap(output['eventhubs']) and values['blob'] == module.unwrap(output['blob'])
          def connections():
              out = phase.read('ephemeral-output.json')
              eh = module.unwrap(out['eventhubs'])['bootstrap_servers'].split(':')[0]
              blob = phase.blob_host()
              ips = module.unwrap(out['private_endpoint_ips'])
              for host, port, ip in ((eh, 9093, ips['eventhubs']), (blob, 443, ips['blob'])):
                  answers = {item[4][0] for item in socket.getaddrinfo(host, port, type=socket.SOCK_STREAM)}
                  assert answers == {ip}
                  with socket.create_connection((ip, port), timeout=15) as connection:
                      assert connection.getpeername()[0] == ip
              return eh, blob, ips
          def pod_connections():
              eh, blob, ips = connections()
              for value in ips.values(): ipaddress.IPv4Address(value)
              assert re.fullmatch(r'[a-z0-9.-]+', eh) and re.fullmatch(r'[a-z0-9.-]+', blob)
              job = os.environ['PARCHWELL_RUN_ID'] + '-smoke'
              rendered = subprocess.run(['python3', 'infra/scripts/lifecycle/command.py', '120', '--', 'helm', 'template', 'parchwell-smoke', 'charts/smoke', '--namespace', 'parchwell', '--set-string', 'jobName=' + job], capture_output=True, check=True).stdout.decode()
              resolve = "resolve() { nslookup \"$1\" 2>/dev/null | awk -v expected=\"$2\" '/^Name:/{answer=1;next} answer && /^Address( [0-9]+)?:/{found=1; if ($NF != expected) bad=1} END {exit !(found && !bad)}'; }; "
              command = resolve + 'resolve ' + eh + ' ' + ips['eventhubs'] + ' && resolve ' + blob + ' ' + ips['blob'] + ' && nc -z -w 15 ' + ips['eventhubs'] + ' 9093 >/dev/null 2>&1 && nc -z -w 15 ' + ips['blob'] + ' 443 >/dev/null 2>&1 && printf "parchwell-cluster-smoke-ok\\n"'
              lines = rendered.splitlines()
              matches = [i for i, line in enumerate(lines) if line.strip().startswith('command:')]
              assert len(matches) == 1
              lines[matches[0]] = '          command: ' + json.dumps(['/bin/sh', '-c', command])
              manifest = phase.private / 'connection-smoke.yaml'
              manifest.write_text('\n'.join(lines) + '\n')
              def run(args, seconds):
                  subprocess.run(['python3', 'infra/scripts/lifecycle/command.py', str(seconds), '--', *args], check=True)
              try:
                  run(['kubectl', 'apply', '-f', str(manifest), '--request-timeout=30s'], 60)
                  run(['bash', 'infra/scripts/lifecycle/lifecycle.sh', 'wait-job', job, '120'], 210)
                  logs = phase.private / 'connection-smoke.log'
                  run(['python3', 'infra/scripts/lifecycle/command.py', '30', '--stdout-file', str(logs), '--max-bytes', '65536', '--', 'kubectl', 'logs', '-n', 'parchwell', 'job/' + job, '--timestamps=false', '--request-timeout=20s'], 35)
                  assert logs.read_bytes() == b'parchwell-cluster-smoke-ok\n'
              finally:
                  run(['kubectl', 'delete', 'job', job, '-n', 'parchwell', '--ignore-not-found', '--cascade=foreground', '--wait=true', '--timeout=90s', '--request-timeout=30s'], 120)
          def one(mode):
              record = phase.run_mode(mode, 1)
              if record['outcome'] != 'complete' or record.get('collect_gap'):
                  raise RuntimeError('diagnostic worker failed')
          def confirm_loadgen_delivery():
              selected = phase.read('phase.json')
              job = phase.mode_job(selected, 'loadgen', 1)
              result = subprocess.run(['python3', 'infra/scripts/lifecycle/command.py', '30', '--', 'kubectl', 'get', 'pods', '-n', 'parchwell', '-l', 'job-name=' + job, '-o', 'json', '--request-timeout=20s'], capture_output=True, check=True)

              assert re.fullmatch(r'[a-z0-9](?:[a-z0-9-]*[a-z0-9])?', pod)
              name = selected['run_id'] + '/' + selected['phase'] + '/r1/producer/' + pod + '/publication/ledger.jsonl'
              ledger = phase.download_blob(name, phase.private / ('delivery-' + selected['phase'] + '.jsonl'), 268435456, time.monotonic() + 180, container='ledger')
              assert ledger is not None and ledger.endswith(b'\n')
              delivered = False
              for line in io.BytesIO(ledger):
                  row = json.loads(line)
                  identity = row['identity']
                  assert identity['scenario_id'] == 'B04' and identity['configuration_id'] == selected['configuration'] and identity['run_id'] == selected['run_id'] and identity['instance_id'] == pod
                  if row['delivered'] is True:
                      delivered = True
              assert delivered
          def probe_while_loadgen_active():
              import concurrent.futures, datetime
              selected = phase.read('phase.json')
              job = phase.mode_job(selected, 'loadgen', 1)
              def active():
                  result = subprocess.run(['python3', 'infra/scripts/lifecycle/command.py', '30', '--', 'kubectl', 'get', 'job', job, '-n', 'parchwell', '-o', 'json', '--request-timeout=20s'], capture_output=True)
                  if result.returncode: return False
                  return json.loads(result.stdout).get('status', {}).get('active', 0) > 0
              with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
                  future = pool.submit(one, 'loadgen')
                  deadline = time.monotonic() + 120
                  while time.monotonic() < deadline and not active():
                      if future.done(): raise RuntimeError('loadgen ended before connection probes')
                      time.sleep(1)
                  if not active(): raise RuntimeError('loadgen job did not become active')
                  started = datetime.datetime.now(datetime.timezone.utc).isoformat()
                  pod_connections()
                  ended = datetime.datetime.now(datetime.timezone.utc).isoformat()
                  if not active(): raise RuntimeError('loadgen job ended during connection probes')
                  future.result()
              (pathlib.Path(os.environ['RUNNER_TEMP']) / ('configuration-' + selected['phase'].lower() + '-active-window')).write_text(started + ' ' + ended)
          pod_connections()
          one('start-line')
          probe_while_loadgen_active()
          confirm_loadgen_delivery()
          one('verify')
          print('fresh runner and in-cluster idle and workload-active probes and positive in-cluster synthetic delivery passed')
          sentinel = phase.download_blob(os.environ['PARCHWELL_RUN_ID'] + '/C0/r1/start-line.json', phase.private / 'sentinel-before', 16777216, time.monotonic() + 180)
          if sentinel is None:
              raise RuntimeError('sentinel read failed')
          import hashlib
          (phase.private / 'sentinel-sha256').write_text(hashlib.sha256(sentinel).hexdigest())
          import datetime
          (pathlib.Path(os.environ['RUNNER_TEMP']) / 'configuration-c0-time').write_text(datetime.datetime.now(datetime.timezone.utc).isoformat())
          print('C0 fresh runner and in-cluster idle and workload-active probes and positive in-cluster synthetic delivery passed')
          PYTHON

                  if (item.get('workspaceId', '').casefold() == state['workspace_resource_id'].casefold() and item.get('logAnalyticsDestinationType') == 'Dedicated' and logs == wanted_logs and metrics == wanted_metrics): matches.append(item)
