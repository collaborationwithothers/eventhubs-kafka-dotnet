              destination = directory / ('-'.join((phase_name, context, perspective, service, operation)) + '.json')

              for service, host, port, ip in (('eventhubs', eh, 9093, ips['eventhubs']), ('blob', blob, 443, ips['blob'])):
                  try:
                      answers = {item[4][0] for item in socket.getaddrinfo(host, port, type=socket.SOCK_STREAM)}
                  except Exception:
                      save_probe(phase_name, context, 'runner', service, 'dns', 'failed', 'dns_error')
                      raise RuntimeError('runner DNS probe failed') from None
                  if answers != {ip}:
                      save_probe(phase_name, context, 'runner', service, 'dns', 'failed', 'dns_mismatch')
                      raise RuntimeError('runner DNS probe failed')
                  save_probe(phase_name, context, 'runner', service, 'dns', 'passed')
                  try:
                      with socket.create_connection((ip, port), timeout=15) as connection:
                          peer = connection.getpeername()[0]
                  except Exception as error:
                      timed_out = isinstance(error, socket.timeout)
                      save_probe(phase_name, context, 'runner', service, 'tcp', 'timeout' if timed_out else 'failed', 'timeout' if timed_out else 'connection_error')
                      raise RuntimeError('runner TCP probe failed') from None

                      save_probe(phase_name, context, 'runner', service, 'tcp', 'failed', 'peer_mismatch')
