# Installed at .github/workflows/scenario.yml; committed copy at docs/operations/workflows/scenario.yml.txt; the two are byte-identical (AGENTS.md hard safety rule 8).
name: A01 scenario
on:
  workflow_dispatch:
permissions:
  contents: read
  id-token: write
concurrency:
  group: parchwell-ephemeral
  cancel-in-progress: false
jobs:
  probe:
    runs-on: ${{ vars.PARCHWELL_RUNNER_LABEL }}
    environment: parchwell-scenarios
    timeout-minutes: 240
    outputs:
      full_allowed: ${{ steps.gate.outputs.full_allowed }}
      loadgen_digest: ${{ steps.images.outputs.loadgen_digest }}
      verifier_digest: ${{ steps.images.outputs.verifier_digest }}
    env:
      AZURE_CLIENT_ID: ${{ vars.AZURE_CLIENT_ID }}
      AZURE_TENANT_ID: ${{ vars.AZURE_TENANT_ID }}
      AZURE_SUBSCRIPTION_ID: ${{ vars.AZURE_SUBSCRIPTION_ID }}
      ARM_CLIENT_ID: ${{ vars.AZURE_CLIENT_ID }}
      ARM_TENANT_ID: ${{ vars.AZURE_TENANT_ID }}
      ARM_SUBSCRIPTION_ID: ${{ vars.AZURE_SUBSCRIPTION_ID }}

      - name: Require the main branch before login
        shell: bash
        run: |
          set -euo pipefail
          test "$GITHUB_REF" = refs/heads/main
          for value in "$AZURE_CLIENT_ID" "$AZURE_TENANT_ID" "$AZURE_SUBSCRIPTION_ID"; do
            [[ "$value" =~ ^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$ ]]
            echo "::add-mask::$value"
          done
      - name: Check out the main head this run started from
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.sha }}
          persist-credentials: false
      - name: Install the pinned Terraform
        uses: hashicorp/setup-terraform@dfe3c3f87815947d99a8997f908cb6525fc44e9e # v4.0.1

          terraform_version: 1.16.6

        uses: actions/setup-dotnet@a98b56852c35b8e3190ac28c8c2271da59106c68 # v6.0.0
        with:
          global-json-file: global.json
      - name: Require workflow commands
        shell: bash
        run: |
          set -euo pipefail
          required_commands=(az cat chmod curl dirname dotnet grep kubectl mkdir mktemp python3 terraform timeout)
          missing_commands=()
          for command_name in "${required_commands[@]}"; do
            if ! command -v "$command_name" > /dev/null 2>&1; then
              missing_commands+=("$command_name")
            fi
          done
          if ((${#missing_commands[@]} > 0)); then
            printf 'Missing required commands:' >&2
            printf ' %s' "${missing_commands[@]}" >&2
            printf '\n' >&2
            exit 1
          fi
      - name: Materialise private inputs from environment secrets
        shell: bash
        env:
          S_PERSISTENT_OUTPUTS: ${{ secrets.PARCHWELL_PERSISTENT_OUTPUTS }}
          S_EPHEMERAL_BACKEND: ${{ secrets.PARCHWELL_EPHEMERAL_BACKEND }}
          S_EPHEMERAL_TFVARS: ${{ secrets.PARCHWELL_EPHEMERAL_TFVARS }}
          S_COST_INPUT: ${{ secrets.PARCHWELL_COST_INPUT }}
          S_READINESS: ${{ secrets.PARCHWELL_RUN_STORAGE_READINESS }}
          S_RUNTIME_PROBE: ${{ secrets.PARCHWELL_RUNTIME_PROBE }}

          set -euo pipefail
          umask 077
          write() { test -n "$3" || { echo "secret $2 is empty" >&2; exit 1; }; mkdir -p "$(dirname "$1")"; printf '%s\n' "$3" > "$1"; }

          write "$PARCHWELL_EPHEMERAL_BACKEND_CONFIG" S_EPHEMERAL_BACKEND "$S_EPHEMERAL_BACKEND"
          write "$PARCHWELL_EPHEMERAL_TFVARS_PATH" S_EPHEMERAL_TFVARS "$S_EPHEMERAL_TFVARS"

          write "$PARCHWELL_RUN_STORAGE_READINESS_PATH" S_READINESS "$S_READINESS"
          write "$PARCHWELL_A01_RUNTIME_DIRECTORY/probe.json" S_RUNTIME_PROBE "$S_RUNTIME_PROBE"

          for key in required:
              assert os.environ.get(key) and pathlib.Path(os.environ[key]).is_file(), 'required_private_input_missing'
          runtime_path = pathlib.Path(os.environ['PARCHWELL_A01_RUNTIME_DIRECTORY']) / (os.environ['A01_PHASE'] + '.json')

              cost_file=os.environ['PARCHWELL_COST_INPUT_PATH'], readiness_file=os.environ['PARCHWELL_RUN_STORAGE_READINESS_PATH'],

              chart_directory=str(pathlib.Path('charts/parchwell').resolve()), limits_file=str(pathlib.Path('docs/operations/a01-runtime-limits.json').resolve()),
              helm_archive_file=str(pathlib.Path(os.environ['A01_OUTPUT'],'tools','helm.tar.gz')),
              helm_binary_file=str(pathlib.Path(os.environ['RUNNER_TEMP'],'a01-bin','helm')),
              verifier_executable='dotnet', verifier_arguments=[str(pathlib.Path('src/Parchwell.Verifier/bin/Release/net10.0/Parchwell.Verifier.dll').resolve())])
          assert isinstance(runtime['initialize_ledger'], bool)
          assert runtime['phase'] == 'probe' or runtime['initialize_ledger'] is False
          pathlib.Path(os.environ['A01_OUTPUT'],'runtime.json').write_text(json.dumps(runtime))
          PYTHON

        shell: bash

          set -euo pipefail

          root = pathlib.Path(os.environ['A01_OUTPUT'],'tools')
          # Official release body: https://api.github.com/repos/helm/helm/releases/tags/v4.3.1
          digest = '0403a88ae20913d11b10a2467b50087721fea7581736da6b9d3992f5357fe54b'
          assert hashlib.sha256(root.joinpath('helm.tar.gz').read_bytes()).hexdigest() == digest
          with tarfile.open(root/'helm.tar.gz') as archive:
              binary = archive.extractfile('linux-amd64/helm').read()
          target = pathlib.Path(os.environ['RUNNER_TEMP'],'a01-bin','helm'); target.write_bytes(binary); target.chmod(0o700)
          root.joinpath('helm.json').write_text(__import__('json').dumps({'version':'v4.3.1','source_url':'https://get.helm.sh/helm-v4.3.1-linux-amd64.tar.gz','checksum_source':'https://api.github.com/repos/helm/helm/releases/tags/v4.3.1','expected_sha256':digest,'actual_sha256':digest,'verified':True,'binary_sha256':hashlib.sha256(binary).hexdigest()}))
          PYTHON
          test "$("$RUNNER_TEMP/a01-bin/helm" version --template '{{.Version}}')" = v4.3.1
          echo "$RUNNER_TEMP/a01-bin" >> "$GITHUB_PATH"

          printf 'PARCHWELL_AZ_SESSION_DIRECTORY=%s\n' "$session_dir" >> "$GITHUB_ENV"
          printf '%s\n' "$GITHUB_WORKSPACE/scripts/azure-cli-session" >> "$GITHUB_PATH"

        shell: bash

          set -euo pipefail
          python3 - <<'PYTHON'
          import json, os, pathlib, re, subprocess
          out=pathlib.Path(os.environ['A01_OUTPUT']); runtime=json.loads((out/'runtime.json').read_text())
          acr=runtime['persistent_outputs']['acr']; acr=acr.get('value',acr)
          sha=runtime['source_commit']; registry=acr['login_server']; name=acr['name']
          assert re.fullmatch('[a-z0-9]+',name) and re.fullmatch('[a-z0-9.-]+',registry)
          with (out/'image-publication.log').open('wb') as log:
              subprocess.run(['timeout','60','az','acr','login','--name',name],check=True,stdout=log,stderr=log)
              for worker,project in [('loadgen','LoadGen'),('verifier','Verifier')]:
                  subprocess.run(['timeout','600','dotnet','publish',f'src/Parchwell.{project}/Parchwell.{project}.csproj','--configuration','Release','-p:RestoreLockedMode=true','-p:NuGetLockFilePath=packages.linux-x64.lock.json','-r','linux-x64','--self-contained','false','/t:PublishContainer',f'-p:ContainerRegistry={registry}',f'-p:ContainerImageTag={sha}'],check=True,stdout=log,stderr=log)
                  digest=subprocess.check_output(['timeout','60','az','acr','repository','show','--name',name,'--image',f'parchwell/{worker}:{sha}','--query','digest','--output','tsv'],stderr=log,text=True).strip()
                  assert re.fullmatch('sha256:[0-9a-f]{64}',digest)
                  runtime.setdefault('images',{})[worker]={'source_commit':sha,'digest':digest}
                  with open(os.environ['GITHUB_OUTPUT'],'a') as output: output.write(worker+'_digest='+digest+'\n')

          (out/'image-receipts.json').write_text(json.dumps(runtime['images']))
