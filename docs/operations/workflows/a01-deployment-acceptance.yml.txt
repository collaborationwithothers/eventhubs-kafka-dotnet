# Optional offline acceptance job for Hari to install; no Azure identity or live run.
# This does not replace the existing .NET verification workflow evidence.
name: A01 offline deployment acceptance
on:
  workflow_dispatch:
permissions:
  contents: read
jobs:
  a01-deployment-acceptance:
    runs-on: ubuntu-24.04
    timeout-minutes: 45
    steps:
      - name: Check out tested source
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.sha }}
          persist-credentials: false
      - name: Set up pinned SDK
        uses: actions/setup-dotnet@a98b56852c35b8e3190ac28c8c2271da59106c68 # v6.0.0
        with:
          global-json-file: global.json
      - name: Prepare offline tools and exact source build
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p "$RUNNER_TEMP/a01-tools" "$RUNNER_TEMP/a01-deployment-evidence"
          # Official Helm v4.3.1 release lists this linux-amd64 archive checksum.
          # Source: https://api.github.com/repos/helm/helm/releases/tags/v4.3.1
          curl --fail --location --max-time 60 --max-filesize 67108864 https://get.helm.sh/helm-v4.3.1-linux-amd64.tar.gz -o "$RUNNER_TEMP/a01-tools/helm.tar.gz"
          python3 - <<'PYTHON'
          import hashlib,os,pathlib
          path=pathlib.Path(os.environ['RUNNER_TEMP'],'a01-tools','helm.tar.gz')
          assert hashlib.sha256(path.read_bytes()).hexdigest()=='0403a88ae20913d11b10a2467b50087721fea7581736da6b9d3992f5357fe54b'
          PYTHON
          echo "PARCHWELL_HELM_ARCHIVE=$RUNNER_TEMP/a01-tools/helm.tar.gz" >> "$GITHUB_ENV"
          timeout 60 python3 -c 'import yaml' # PyYAML must be provisioned; no floating install.
          timeout 60 docker info > "$RUNNER_TEMP/a01-deployment-evidence/docker-info.log"
          timeout 600 dotnet restore Parchwell.slnx --locked-mode > "$RUNNER_TEMP/a01-deployment-evidence/restore.log" 2>&1
          timeout 600 dotnet build Parchwell.slnx --configuration Release --no-restore > "$RUNNER_TEMP/a01-deployment-evidence/build.log" 2>&1
      - name: Exercise real publication, smoke, generation and Helm assertions
        shell: bash
        run: |
          timeout 600 ./scripts/verify-a01-deployment.sh --output "$RUNNER_TEMP/a01-deployment-evidence"
      - name: Rehearse with fresh real-writer fixture bytes
        shell: bash
        run: |
          timeout 600 dotnet run --project src/Parchwell.Runner/Parchwell.Runner.csproj --configuration Release --no-build -- rehearse --fixtures "$RUNNER_TEMP/a01-deployment-evidence/wire-fixtures" --output "$RUNNER_TEMP/a01-rehearsal"
      - name: Upload synthetic offline acceptance evidence
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: a01-offline-deployment-acceptance
          path: |
            ${{ runner.temp }}/a01-deployment-evidence
            ${{ runner.temp }}/a01-rehearsal
          if-no-files-found: error
